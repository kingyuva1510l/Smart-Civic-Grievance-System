trigger ComplaintTrigger on Complaint__c (before insert, after insert) {

    if(Trigger.isBefore){
        if(Trigger.isInsert){
            ComplaintHandler.applySLAPolicy(Trigger.new);
        }
    }

    if(Trigger.isAfter){
        if(Trigger.isInsert){
            ComplaintHandler.sendEmailNotification(Trigger.new);
        }
    }

}
